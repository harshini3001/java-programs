import java.util.Scanner;

public class Main {

    public static void main(String args[]) {
        Scanner x = new Scanner(System.in);

        String s = x.nextLine();
        countWords(s);
    }

    static void countWords(String s) {

        int count = 0;

        for (int i = 0; i < s.length(); i++) {

            if (s.charAt(i) != ' ' && (i == 0 || s.charAt(i - 1) == ' ')) {
                count++;
            }
        }

        System.out.print(count);
    }
}
