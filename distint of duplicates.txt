import java.util.Scanner;
public class Main{ 
    public static void duplicates(int arr[],int n){
        boolean visited[]=new boolean[n];
        for(int i=0;i<n;i++){
            if(visited[i]){
                continue;
            }
            int count=1;
            for(int j=i+1;j<n;j++){
                if(arr[i]==arr[j]){
                    count++;
                    visited[j]=true;
                }
            }
                if(count==1){
                    System.out.print(arr[i] + " ");
                    
                }
        }
    }
	public static void main(String[] args) {
	    Scanner s=new Scanner(System.in);
	    int n =s.nextInt();
	    int arr[]=new int[n];
	   for(int i=0;i<n;i++){
	       arr[i]=s.nextInt();
	   }
	   duplicates(arr,n);
	}
    
	
}
