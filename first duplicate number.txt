import java.util.Scanner;
public class Main{ 
    public static void firstDuplicate(int arr[],int n){
        boolean found=false;
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                if(arr[i]==arr[j]){
                    System.out.println(arr[i]);
                }   found=true;
            }       break;
        }
    }
	public static void main(String[] args) {
	    Scanner s=new Scanner(System.in);
	    int n =s.nextInt();
	    int arr[]=new int[n];
	   for(int i=0;i<n;i++){
	       arr[i]=s.nextInt();
	   }
	   firstDuplicate(arr,n);
	}
}
