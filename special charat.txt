import java.util.Scanner;

public class Main {

    public static void main(String args[]) {
        Scanner x = new Scanner(System.in);
        String s = x.nextLine();
        printspecial(s);
    }

    static void printspecial(String s) {
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if (!((ch >= 'A' && ch <= 'Z') ||
                  (ch >= 'a' && ch <= 'z') ||
                  (ch >= '0' && ch <= '9'))) {
                System.out.print(ch);
            }
        }
    }
}
